// 📁 js/engine/pipeline-orchestrator.js

class PipelineOrchestrator {
  async processProject(projectPath, options = {}) {
    const result = {
      success: false,
      mermaidCode: '',
      warnings: [],
      fallbackUsed: false
    };
    
    try {
      // 階段 1: 程式分析
      const analysisResult = await this.analyzeProject(projectPath);
      if (!analysisResult || !analysisResult.entities.length) {
        result.warnings.push('Analysis produced no entities');
      }
      
      // 階段 2: IR 轉換
      const ir = this.convertToIR(analysisResult);
      if (!ir || !ir.isValid()) {
        result.warnings.push('IR conversion failed, using minimal structure');
      }
      
      // 階段 3: Mermaid 生成
      const mermaidCode = this.generateMermaid(ir, options);
      if (!mermaidCode || mermaidCode.length === 0) {
        // 最小可用輸出
        result.mermaidCode = this.generateMinimalDiagram(projectPath);
        result.fallbackUsed = true;
      } else {
        result.mermaidCode = mermaidCode;
      }
      
      // 階段 4: AutoFix 處理
      if (result.mermaidCode) {
        result.mermaidCode = await this.applyAutoFix(result.mermaidCode);
        result.success = true;
      }
      
    } catch (error) {
      result.warnings.push(`Pipeline error: ${error.message}`);
      result.mermaidCode = this.generateErrorDiagram(error);
      result.fallbackUsed = true;
    }
    
    return result;
  }
  
  generateMinimalDiagram(projectPath) {
    // 最小可用圖表，確保有輸出
    return `
flowchart TD
    A[Project: ${path.basename(projectPath)}]
    A --> B[Analysis Incomplete]
    B --> C[Using Fallback Mode]
    
    classDef warning fill:#fff2cc,stroke:#d6b656
    class B,C warning
`;
  }
}

import { EventEmitter } from 'eventemitter3'
import { fromEvent, Observable } from 'rxjs'

export interface SystemEvents {
  // Pipeline 事件
  'pipeline:started': { projectId: string; files: string[] }
  'pipeline:progress': { projectId: string; progress: number }
  'pipeline:completed': { projectId: string; result: any }
  
  // 錯誤事件
  'error:recoverable': { error: Error; context: any; recovery?: string }
  'error:fatal': { error: Error; context: any }
  
  // 系統事件
  'system:memory:warning': { usage: number; threshold: number }
  'system:worker:status': { workerId: string; status: string }
}

export class ModernEventBus extends EventEmitter<SystemEvents> {
  // 響應式事件流
  pipeline$ = fromEvent(this as any, /pipeline:.*/)
  errors$ = fromEvent(this as any, /error:.*/)
  system$ = fromEvent(this as any, /system:.*/)
  
  // 型別安全的事件發射
  emitPipelineStarted(data: SystemEvents['pipeline:started']) {
    this.emit('pipeline:started', data)
    this.broadcastToDOM('pipeline:started', data)
  }
  
  emitError(type: 'recoverable' | 'fatal', error: Error, context: any) {
    this.emit(`error:${type}` as any, { error, context })
    this.broadcastToDOM(`error:${type}`, { error: error.message, context })
  }
  
  private broadcastToDOM(type: string, data: any) {
    if (typeof window !== 'undefined') {
      window.dispatchEvent(
        new CustomEvent(`autofix:${type}`, { detail: data })
      )
    }
  }
}

export const globalEventBus = new ModernEventBus()

import { globalEventBus } from '../modern/events/EventBus.js'

export class EventIntegratedPipeline {
  constructor(originalPipeline) {
    this.pipeline = originalPipeline
    this.setupEventHandlers()
  }
  
  async processProject(files, options) {
    const projectId = options.projectId || `project-${Date.now()}`
    
    try {
      globalEventBus.emitPipelineStarted({ projectId, files })
      
      // 使用現有 pipeline，增加事件發射
      const result = await this.pipeline.processProject(files, options)
      
      globalEventBus.emit('pipeline:completed', { projectId, result })
      return result
      
    } catch (error) {
      globalEventBus.emitError('recoverable', error, { files, options })
      throw error
    }
  }
  
  setupEventHandlers() {
    // 連接現有事件到新事件系統
    if (this.pipeline.on) {
      this.pipeline.on('progress', (progress) => {
        globalEventBus.emit('pipeline:progress', {
          projectId: 'current',
          progress: progress.percentage || 0
        })
      })
    }
  }
}

#### **5.1 統一錯誤管理**
```typescript
// 📁 modern/errors/ErrorManager.ts

export class ModernErrorManager {
  private recoveryStrategies = new Map<string, RecoveryStrategy>()
  
  constructor() {
    this.registerDefaultStrategies()
    this.setupEventHandlers()
  }
  
  async handleError(error: Error, context: any) {
    const errorType = this.classifyError(error)
    const strategy = this.recoveryStrategies.get(errorType)
    
    if (strategy && strategy.canRecover(error, context)) {
      try {
        const result = await strategy.recover(error, context)
        globalEventBus.emitError('recoverable', error, { 
          ...context, 
          recovery: strategy.name,
          result 
        })
        return result
      } catch (recoveryError) {
        globalEventBus.emitError('fatal', recoveryError, context)
        throw recoveryError
      }
    }
    
    globalEventBus.emitError('fatal', error, context)
    throw error
  }
  
  private registerDefaultStrategies() {
    // Tree-sitter 失敗恢復
    this.recoveryStrategies.set('TreeSitterError', {
      name: 'FallbackToRegex',
      canRecover: (error) => error.message.includes('TreeSitter'),
      recover: async (error, context) => {
        console.warn('Tree-sitter failed, using regex fallback')
        return this.fallbackParser.parse(context.code, context.language)
      }
    })
    
    // 記憶體不足恢復
    this.recoveryStrategies.set('MemoryError', {
      name: 'MemoryCleanup',
      canRecover: (error) => error.message.includes('memory'),
      recover: async (error, context) => {
        await this.memoryManager.cleanup()
        globalEventBus.emit('system:memory:warning', {
          usage: process.memoryUsage().heapUsed,
          threshold: 500 * 1024 * 1024 // 500MB
        })
        // 重試處理
        return this.retryWithCleanup(context)
      }
    })
  }
}
```

// 📁 libs/core/src/pipeline/ProcessingPipeline.ts

export interface PipelineConfig {
  maxConcurrent: number
  enableMetrics: boolean
  timeout: number
  retryAttempts: number
}

export interface ProcessingResult<T = any> {
  success: boolean
  data: T
  errors: Error[]
  metrics: ProcessingMetrics
  fallbackUsed: boolean
}

export class ModernProcessingPipeline<TInput, TOutput> {
  constructor(
    private config: PipelineConfig,
    private processor: (input: TInput) => Promise<TOutput>
  ) {}
  
  async process(input: TInput): Promise<ProcessingResult<TOutput>> {
    const startTime = performance.now()
    const result: ProcessingResult<TOutput> = {
      success: false,
      data: null as any,
      errors: [],
      metrics: { duration: 0, memoryUsage: 0 },
      fallbackUsed: false
    }
    
    try {
      result.data = await this.processor(input)
      result.success = true
    } catch (error) {
      result.errors.push(error as Error)
      // 嘗試恢復
      const recovered = await this.attemptRecovery(input, error as Error)
      if (recovered) {
        result.data = recovered
        result.success = true
        result.fallbackUsed = true
      }
    } finally {
      result.metrics.duration = performance.now() - startTime
      result.metrics.memoryUsage = this.getMemoryUsage()
    }
    
    return result
  }
}

export class PluggableRuleEngine {

  async applyRules(input: string, context: RuleContext): Promise<string> {
    let result = input
    const applicableRules = this.getApplicableRules(context)
    
    for (const rule of applicableRules) {
      try {
        const ruleResult = rule.execute({ ...context, input: result })
        if (ruleResult.success) {
          result = ruleResult.output
        }
      } catch (error) {
        console.warn(`Rule ${rule.id} failed:`, error)
      }
    }
    
    return result
  }
}
